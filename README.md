Build and Test

6\. Kubernetes Deployment Evidence



Minikube provides a local Kubernetes environment

Deployment YAML



&#x20;deployment.yaml defines a Deployment named nginx-deployment

apiVersion: apps/v1

kind: Deployment

metadata:

&#x20; name: nginx-deployment

spec:

&#x20; replicas: 2

&#x20; selector:

&#x20;   matchLabels:

&#x20;     app: nginx

&#x20; template:

&#x20;   metadata:

&#x20;     labels:

&#x20;       app: nginx

&#x20;   spec:

&#x20;     containers:

&#x20;       - name: nginx

&#x20;         image: nginx:1.27

&#x20;         ports:

&#x20;           - containerPort: 80



Service YAML



service.yaml defines a NodePort 

apiVersion: v1

kind: Service

metadata:

&#x20; name: nginx-service

spec:

&#x20; type: NodePort

&#x20; selector:

&#x20;   app: nginx

&#x20; ports:

&#x20;   - port: 80

&#x20;     targetPort: 80

&#x20;     nodePort: 30080
